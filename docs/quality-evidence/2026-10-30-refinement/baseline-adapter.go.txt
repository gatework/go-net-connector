package netdisco
import (
 "context"
 "github.com/gatework/go-net-connector/inventory"
 "github.com/jackc/pgx/v5/pgconn"
)
func decodeHTTPInventory(_ context.Context, body []byte, maximum int, count *int) ([]inventory.Device,error) { return legacyDecodeHTTPInventory(body,maximum,count) }
func postgresColumns(_ context.Context, fields []pgconn.FieldDescription) ([]pgconn.FieldDescription,error) {
 seen:=make(map[string]bool); valid:=false
 for _,f:=range fields { if seen[f.Name] {return nil,invError("invalid_inventory",nil)}; seen[f.Name]=true; if f.Name=="ip" {valid=true} }
 if !valid {return nil,invError("invalid_inventory",nil)}
 return fields,nil
}
func decodePostgresRow(_ context.Context, fields []pgconn.FieldDescription, values [][]byte, limits Limits, response,total *int64) (inventory.Device,error) {return legacyDecodePostgresRow(fields,values,limits,response,total)}
